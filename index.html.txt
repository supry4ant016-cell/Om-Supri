  <script>
    // GANTI DENGAN URL WEB APP APPS SCRIPT (/exec) ANDA
    const APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbxrrTbWHAnT43rVwXsZSDfo06aqQpmbfXYV_PHfqxzVp0kKgC8IDS-HMbjZSU2nTbhQ/exec';

    let allProducts = [];
    let storeSettings = { nama_toko: 'Toko Digital', bio_toko: '', no_wa: '6281234567890' };
    let activeCategory = 'ALL';
    let cart = [];

    async function initStore() {
      try {
        const res = await fetch(`${APPS_SCRIPT_URL}?action=getStore`);
        const json = await res.json();
        if (json.status === 'success') {
          allProducts = json.products || [];
          storeSettings = json.settings || storeSettings;
          document.getElementById('store-name').innerText = storeSettings.nama_toko;
          document.getElementById('store-bio').innerText = storeSettings.bio_toko;
          renderCategories();
          renderProducts(allProducts);
        }
      } catch (e) {
        document.getElementById('product-grid').innerHTML = `<div class="col-span-full text-center py-12 text-xs text-rose-500">Gagal memuat toko. Periksa URL Apps Script Anda.</div>`;
      }
    }

    function renderCategories() {
      const cats = ['ALL', ...new Set(allProducts.map(p => p.kategori).filter(Boolean))];
      document.getElementById('category-tabs').innerHTML = cats.map(c => `
        <button onclick="setKategori('${c}', this)" class="cat-btn ${c === activeCategory ? 'bg-slate-900 text-white' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'} px-4 py-1.5 rounded-full text-xs font-bold shrink-0 transition">
          ${c === 'ALL' ? 'Semua' : c}
        </button>
      `).join('');
    }

    function setKategori(cat, el) {
      activeCategory = cat;
      renderCategories();
      filterKatalog();
    }

    function filterKatalog() {
      const q = document.getElementById('search-input').value.toLowerCase();
      const filtered = allProducts.filter(p => {
        const matchCat = activeCategory === 'ALL' || p.kategori === activeCategory;
        const matchText = p.nama_produk.toLowerCase().includes(q) || (p.deskripsi || '').toLowerCase().includes(q);
        return matchCat && matchText;
      });
      renderProducts(filtered);
    }

    function renderProducts(list) {
      const grid = document.getElementById('product-grid');
      if (list.length === 0) {
        grid.innerHTML = `<div class="col-span-full text-center py-12 text-xs text-slate-400">Produk belum tersedia.</div>`;
        return;
      }
      grid.innerHTML = list.map(p => `
        <div class="bg-white rounded-2xl border border-slate-200/80 overflow-hidden flex flex-col justify-between hover:shadow-lg transition">
          <div>
            <div class="relative aspect-[4/3] bg-slate-100 cursor-pointer" onclick="bukaPreview('${p.id_produk}')">
              <img src="${p.gambar_url || 'https://placehold.co/600x450?text=Produk+Digital'}" class="w-full h-full object-cover">
              <span class="absolute top-2.5 left-2.5 bg-white/95 text-slate-800 text-[10px] font-extrabold px-2.5 py-1 rounded-full shadow-xs">${p.kategori}</span>
              <span class="absolute bottom-2 right-2 bg-slate-900/80 text-white text-[10px] font-bold px-2.5 py-1 rounded-lg">👁️ Intip Isi</span>
            </div>
            <div class="p-3.5">
              <h2 onclick="bukaPreview('${p.id_produk}')" class="font-bold text-xs sm:text-sm text-slate-900 line-clamp-2 cursor-pointer hover:text-violet-600">${p.nama_produk}</h2>
              <p class="text-sm sm:text-base font-extrabold text-violet-600 mt-1.5">Rp ${Number(p.harga).toLocaleString('id-ID')}</p>
            </div>
          </div>
          <div class="p-3.5 pt-0 flex gap-1.5">
            <button onclick="tambahKeranjang('${p.id_produk}')" class="bg-slate-100 hover:bg-slate-200 text-slate-800 px-3 py-2 rounded-xl text-xs font-extrabold transition" title="Simpan ke Keranjang">
              + 🛒
            </button>
            <a href="${p.link_co || '#'}" target="_blank" class="flex-1 bg-slate-900 hover:bg-violet-600 text-white py-2 rounded-xl text-xs font-extrabold text-center transition">
              Beli Langsung
            </a>
          </div>
        </div>
      `).join('');
    }

    function bukaPreview(id) {
      const p = allProducts.find(x => x.id_produk === id);
      if (!p) return;
      document.getElementById('prev-cat').innerText = p.kategori;
      document.getElementById('prev-title').innerText = p.nama_produk;
      document.getElementById('prev-img').src = p.gambar_url || 'https://placehold.co/600x450?text=Produk+Digital';
      document.getElementById('prev-desc').innerText = p.deskripsi || '';
      document.getElementById('prev-price').innerText = 'Rp ' + Number(p.harga).toLocaleString('id-ID');

      const items = (p.isi_produk || 'Akses file digital lengkap').split('|');
      document.getElementById('prev-list').innerHTML = items.map(i => `<li class="flex items-start gap-2"><span class="text-emerald-500 font-bold">✓</span><span>${i.trim()}</span></li>`).join('');

      const sampleBtn = document.getElementById('prev-sample-btn');
      if (p.preview_url) {
        sampleBtn.href = p.preview_url;
        sampleBtn.classList.remove('hidden');
      } else {
        sampleBtn.classList.add('hidden');
      }

      document.getElementById('prev-add-cart').onclick = () => { tambahKeranjang(p.id_produk); tutupModal('modal-preview'); };
      document.getElementById('prev-buy-now').href = p.link_co || '#';
      document.getElementById('modal-preview').classList.remove('hidden');
    }

    function tambahKeranjang(id) {
      const p = allProducts.find(x => x.id_produk === id);
      if (!p) return;
      if (!cart.find(c => c.id_produk === id)) {
        cart.push(p);
      }
      updateCartBar();
    }

    function hapusKeranjang(id) {
      cart = cart.filter(c => c.id_produk !== id);
      updateCartBar();
      renderCartItems();
      if (cart.length === 0) tutupModal('modal-cart');
    }

    function updateCartBar() {
      const bar = document.getElementById('floating-cart');
      if (cart.length === 0) return bar.classList.add('hidden');
      bar.classList.remove('hidden');
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      document.getElementById('cart-badge').innerText = cart.length;
      document.getElementById('cart-total-bar').innerText = 'Rp ' + total.toLocaleString('id-ID');
    }

    function bukaKeranjang() {
      renderCartItems();
      document.getElementById('modal-cart').classList.remove('hidden');
    }

    function renderCartItems() {
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      document.getElementById('cart-modal-total').innerText = 'Rp ' + total.toLocaleString('id-ID');
      document.getElementById('cart-items-list').innerHTML = cart.map(item => `
        <div class="flex items-center justify-between gap-2 bg-slate-50 p-3 rounded-2xl border border-slate-200/70">
          <div class="min-w-0 flex-1">
            <p class="text-xs font-bold text-slate-800 truncate">${item.nama_produk}</p>
            <p class="text-xs font-extrabold text-violet-600">Rp ${Number(item.harga).toLocaleString('id-ID')}</p>
          </div>
          <div class="flex items-center gap-1.5 shrink-0">
            <a href="${item.link_co || '#'}" target="_blank" class="bg-violet-600 hover:bg-violet-700 text-white px-3 py-1.5 rounded-xl text-[11px] font-extrabold">
              Bayar / CO &nearr;
            </a>
            <button onclick="hapusKeranjang('${item.id_produk}')" class="w-7 h-7 rounded-xl bg-rose-100 text-rose-600 text-xs font-bold">&times;</button>
          </div>
        </div>
      `).join('');
    }

    function checkoutViaWA() {
      const daftar = cart.map((item, idx) => `${idx + 1}. ${item.nama_produk} (Rp ${Number(item.harga).toLocaleString('id-ID')})`).join('%0A');
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      const teks = `Halo Admin ${storeSettings.nama_toko}, saya mau order produk digital berikut:%0A%0A${daftar}%0A%0ATotal: Rp ${total.toLocaleString('id-ID')}`;
      window.open(`https://wa.me/${storeSettings.no_wa}?text=${teks}`, '_blank');
    }

    function tutupModal(id) {
      document.getElementById(id).classList.add('hidden');
    }

    initStore();
  </script>

  <script>
    // GANTI DENGAN URL WEB APP APPS SCRIPT (/exec) ANDA
    const APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbxrrTbWHAnT43rVwXsZSDfo06aqQpmbfXYV_PHfqxzVp0kKgC8IDS-HMbjZSU2nTbhQ/exec';

    let products = [];
    let storeSettings = { nama_toko: 'Om Supri Studio', bio_toko: '', no_wa: '6281234567890' };
    let activeCategory = 'ALL';
    let searchQuery = '';
    let cart = [];

    async function initStore() {
      try {
        const res = await fetch(`${APPS_SCRIPT_URL}?action=getStore`);
        const json = await res.json();
        if (json.status === 'success') {
          products = json.products || [];
          storeSettings = json.settings || storeSettings;
          document.getElementById('store-title-nav').innerText = storeSettings.nama_toko;
          document.getElementById('store-title-footer').innerText = storeSettings.nama_toko;
          if (storeSettings.bio_toko) document.getElementById('store-bio-hero').innerText = storeSettings.bio_toko;
          renderCategories();
          filterProducts();
        }
      } catch (e) {
        document.getElementById('product-grid').innerHTML = `<div class="col-span-full bg-white rounded-3xl p-12 text-center card-elevated text-xs text-rose-500">Gagal memuat toko. Pastikan APPS_SCRIPT_URL sudah diisi dengan benar.</div>`;
      }
    }

    function syncSearch(val) {
      searchQuery = val.toLowerCase();
      document.getElementById('search-desktop').value = val;
      document.getElementById('search-mobile').value = val;
      filterProducts();
    }

    function renderCategories() {
      const cats = ['ALL', ...new Set(products.map(p => p.kategori).filter(Boolean))];
      document.getElementById('category-tabs').innerHTML = cats.map(c => `
        <button onclick="setCategory('${c}')" class="${c === activeCategory ? 'bg-zinc-950 text-white shadow-xs' : 'bg-white text-zinc-600 border border-zinc-200/90 hover:border-zinc-300'} px-4 py-2 rounded-full text-xs font-bold shrink-0 transition">
          ${c === 'ALL' ? 'Semua Koleksi' : c}
        </button>
      `).join('');
    }

    function setCategory(c) {
      activeCategory = c;
      renderCategories();
      filterProducts();
    }

    function filterProducts() {
      const sort = document.getElementById('sort-select').value;
      let list = products.filter(p => {
        const matchCat = activeCategory === 'ALL' || p.kategori === activeCategory;
        const matchSearch = String(p.nama_produk).toLowerCase().includes(searchQuery) || String(p.deskripsi || '').toLowerCase().includes(searchQuery);
        return matchCat && matchSearch;
      });
      if (sort === 'low') list.sort((a, b) => Number(a.harga) - Number(b.harga));
      if (sort === 'high') list.sort((a, b) => Number(b.harga) - Number(a.harga));

      document.getElementById('product-count').innerText = list.length;
      renderGrid(list);
    }

    function renderMediaElement(p, extraClass = '') {
      if (p.media_type === 'video' && p.media_url) {
        return `<video src="${p.media_url}" class="w-full h-full object-cover ${extraClass}" autoplay muted loop playsinline></video>`;
      }
      return `<img src="${p.media_url || 'https://placehold.co/600x450/18181b/ffffff?text=Digital+Asset'}" class="w-full h-full object-cover ${extraClass}">`;
    }

    function renderGrid(list) {
      const grid = document.getElementById('product-grid');
      if (list.length === 0) {
        grid.innerHTML = `<div class="col-span-full bg-white rounded-3xl p-12 text-center card-elevated text-xs text-zinc-400">Belum ada produk pada kategori ini.</div>`;
        return;
      }
      grid.innerHTML = list.map(p => {
        const harga = Number(p.harga);
        const oldPrice = Math.round(harga * 1.4);
        return `
          <div class="bg-white rounded-[22px] overflow-hidden flex flex-col justify-between card-elevated hover:-translate-y-1 transition duration-300 group">
            <div>
              <div class="relative aspect-[4/3] bg-zinc-900 cursor-pointer overflow-hidden" onclick="openPreview('${p.id_produk}')">
                ${renderMediaElement(p, 'group-hover:scale-105 transition duration-500')}
                <span class="absolute top-3 left-3 bg-white/95 backdrop-blur-xs text-zinc-950 text-[10px] font-extrabold px-2.5 py-1 rounded-full shadow-xs">${p.kategori}</span>
                ${p.media_type === 'video' ? `<span class="absolute top-3 right-3 bg-indigo-600 text-white text-[9px] font-extrabold px-2 py-0.5 rounded-full">▶ VIDEO</span>` : ''}
                <span class="absolute bottom-2.5 right-2.5 bg-zinc-950/85 backdrop-blur-xs text-white text-[10px] font-bold px-3 py-1 rounded-full">👁️ Intip Isi</span>
              </div>
              <div class="p-4">
                <div class="flex items-center justify-between text-[10px] font-semibold text-zinc-400 mb-1.5">
                  <span class="text-amber-500 font-bold">★ 4.9</span>
                  <span class="text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-full">⚡ Instan</span>
                </div>
                <h2 onclick="openPreview('${p.id_produk}')" class="font-extrabold text-sm text-zinc-950 line-clamp-2 cursor-pointer group-hover:text-indigo-600 leading-snug transition">${p.nama_produk}</h2>
                <div class="mt-3 flex items-baseline gap-2">
                  <p class="text-base font-extrabold text-zinc-950">Rp ${harga.toLocaleString('id-ID')}</p>
                  <span class="text-[11px] text-zinc-400 line-through">Rp ${oldPrice.toLocaleString('id-ID')}</span>
                </div>
              </div>
            </div>
            <div class="p-4 pt-0 flex gap-2">
              <button onclick="addToCart('${p.id_produk}')" class="bg-zinc-100 hover:bg-zinc-200 text-zinc-900 px-3.5 py-2.5 rounded-full text-xs font-extrabold transition" title="Simpan ke Tas">+ Tas</button>
              <a href="${p.link_co || '#'}" target="_blank" class="flex-1 bg-zinc-950 hover:bg-indigo-600 text-white py-2.5 rounded-full text-xs font-extrabold text-center transition">Beli Langsung</a>
            </div>
          </div>
        `;
      }).join('');
    }

    function openPreview(id) {
      const p = products.find(x => x.id_produk === id);
      if (!p) return;
      const harga = Number(p.harga);
      document.getElementById('prev-cat').innerText = p.kategori;
      document.getElementById('prev-title').innerText = p.nama_produk;
      document.getElementById('prev-media-wrap').innerHTML = renderMediaElement(p);
      document.getElementById('prev-desc').innerText = p.deskripsi || '';
      document.getElementById('prev-old-price').innerText = 'Rp ' + Math.round(harga * 1.4).toLocaleString('id-ID');
      document.getElementById('prev-price').innerText = 'Rp ' + harga.toLocaleString('id-ID');

      const items = (p.isi_produk || 'Akses penuh file produk digital').split('|');
      document.getElementById('prev-list').innerHTML = items.map(i => `<li class="flex items-start gap-2"><span class="text-emerald-600 font-extrabold">✓</span><span>${i.trim()}</span></li>`).join('');

      const sampleBtn = document.getElementById('prev-sample-btn');
      if (p.preview_url) {
        sampleBtn.href = p.preview_url;
        sampleBtn.classList.remove('hidden');
      } else {
        sampleBtn.classList.add('hidden');
      }

      document.getElementById('prev-add-cart').onclick = () => { addToCart(p.id_produk); closeModal('modal-preview'); };
      document.getElementById('prev-buy-now').href = p.link_co || '#';
      document.getElementById('modal-preview').classList.remove('hidden');
    }

    function addToCart(id) {
      const p = products.find(x => x.id_produk === id);
      if (!p) return;
      if (!cart.find(c => c.id_produk === id)) {
        cart.push(p);
        showToast('Disimpan ke Tas Belanja!');
      } else {
        showToast('Produk sudah ada di Tas Belanja');
      }
      updateCartUI();
    }

    function removeFromCart(id) {
      cart = cart.filter(c => c.id_produk !== id);
      updateCartUI();
      renderCartList();
      if (cart.length === 0) closeModal('modal-cart');
    }

    function updateCartUI() {
      const total = cart.reduce((sum, i) => sum + Number(i.harga), 0);
      document.getElementById('nav-cart-count').innerText = cart.length;
      document.getElementById('cart-badge-mob').innerText = cart.length;
      document.getElementById('cart-total-mob').innerText = 'Rp ' + total.toLocaleString('id-ID');
      const mobBar = document.getElementById('floating-cart');
      if (cart.length > 0) mobBar.classList.remove('hidden');
      else mobBar.classList.add('hidden');
    }

    function openCartModal() {
      if (cart.length === 0) return showToast('Tas belanja masih kosong!');
      renderCartList();
      document.getElementById('modal-cart').classList.remove('hidden');
    }

    function renderCartList() {
      const total = cart.reduce((sum, i) => sum + Number(i.harga), 0);
      document.getElementById('cart-modal-total').innerText = 'Rp ' + total.toLocaleString('id-ID');
      document.getElementById('cart-items-list').innerHTML = cart.map(item => `
        <div class="flex items-center justify-between gap-2 bg-zinc-50 p-3 rounded-2xl border border-zinc-200/70">
          <div class="min-w-0 flex-1">
            <p class="text-xs font-bold text-zinc-900 truncate">${item.nama_produk}</p>
            <p class="text-xs font-extrabold text-indigo-600">Rp ${Number(item.harga).toLocaleString('id-ID')}</p>
          </div>
          <div class="flex items-center gap-1.5 shrink-0">
            <a href="${item.link_co || '#'}" target="_blank" class="bg-zinc-950 hover:bg-indigo-600 text-white px-3 py-1.5 rounded-full text-[11px] font-extrabold transition">Bayar &nearr;</a>
            <button onclick="removeFromCart('${item.id_produk}')" class="w-7 h-7 rounded-full bg-rose-100 text-rose-600 text-xs font-bold">&times;</button>
          </div>
        </div>
      `).join('');
    }

    function checkoutViaWA() {
      const daftar = cart.map((item, idx) => `${idx + 1}. ${item.nama_produk} (Rp ${Number(item.harga).toLocaleString('id-ID')})`).join('%0A');
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      const teks = `Halo Admin ${storeSettings.nama_toko}, saya mau order produk digital berikut:%0A%0A${daftar}%0A%0ATotal: Rp ${total.toLocaleString('id-ID')}`;
      window.open(`https://wa.me/${storeSettings.no_wa}?text=${teks}`, '_blank');
    }

    function chatAdminWA() {
      window.open(`https://wa.me/${storeSettings.no_wa}?text=Halo%20Admin%20${encodeURIComponent(storeSettings.nama_toko)},%20saya%20mau%20tanya%20seputar%20produk%20digitalnya.`, '_blank');
    }

    function closeModal(id) {
      document.getElementById(id).classList.add('hidden');
    }

    let toastTimer;
    function showToast(msg) {
      const box = document.getElementById('toast-box');
      document.getElementById('toast-msg').innerText = msg;
      box.classList.remove('hidden');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => box.classList.add('hidden'), 2800);
    }

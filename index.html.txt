    // Manajemen Keranjang
    function tambahKeranjang(id) {
      const produk = katalogProduk.find(p => p.id_produk === id);
      if (!produk) return;
      const existing = keranjang.find(k => k.id_produk === id);
      if (existing) existing.qty += 1;
      else keranjang.push({ ...produk, qty: 1 });
      updateUIKeranjang();
    }

    function ubahQty(id, delta) {
      const idx = keranjang.findIndex(k => k.id_produk === id);
      if (idx === -1) return;
      keranjang[idx].qty += delta;
      if (keranjang[idx].qty <= 0) keranjang.splice(idx, 1);
      updateUIKeranjang();
      renderCheckoutList();
      if (keranjang.length === 0) tutupModal('modal-checkout');
    }

    function updateUIKeranjang() {
      const totalItem = keranjang.reduce((sum, item) => sum + item.qty, 0);
      const totalHarga = keranjang.reduce((sum, item) => sum + (item.harga * item.qty), 0);
      
      document.getElementById('nav-cart-badge').innerText = totalItem;
      const bar = document.getElementById('floating-cart');
      if (keranjang.length === 0) {
        bar.classList.add('hidden');
      } else {
        bar.classList.remove('hidden');
        document.getElementById('cart-badge').innerText = totalItem;
        document.getElementById('cart-total-text').innerText = 'Rp ' + totalHarga.toLocaleString('id-ID');
      }
    }

    function bukaCheckout() {
      if (keranjang.length === 0) {
        tampilkanStatus('info', 'Keranjang Masih Kosong', 'Pilih produk digital yang kamu inginkan terlebih dahulu.');
        return;
      }
      renderCheckoutList();
      document.getElementById('modal-checkout').classList.remove('hidden');
    }

    }

    async function prosesCheckout(e) {
      e.preventDefault();
      const btn = document.getElementById('btn-bayar');
      const teksAsli = btn.innerText;
      btn.disabled = true;
      btn.innerText = 'Menyiapkan QRIS / Pembayaran...';

      const totalBayar = keranjang.reduce((sum, item) => sum + (item.harga * item.qty), 0);
      const emailPembeli = document.getElementById('input-email').value;

      const payload = {
        action: 'createOrder',
        nama: document.getElementById('input-nama').value,
        email: emailPembeli,
        whatsapp: document.getElementById('input-wa').value,
        total_bayar: totalBayar,
        items: keranjang.map(k => ({ id_produk: k.id_produk, nama: k.nama_produk, harga: k.harga, qty: k.qty }))
      };

      try {
        if (APPS_SCRIPT_URL.includes('GANTI_DENGAN_ID')) {
          tutupModal('modal-checkout');
          tampilkanStatus('info', 'Mode Pratinjau UI', 'Hubungkan URL Google Apps Script & Client Key Midtrans Anda agar popup QRIS langsung muncul di sini.');
          return;
        }

        const res = await fetch(APPS_SCRIPT_URL, { method: 'POST', body: JSON.stringify(payload) });
        const data = await res.json();

        if (data.status === 'success' && data.snap_token) {
          tutupModal('modal-checkout');
          window.snap.pay(data.snap_token, {
            onSuccess: function() {
              keranjang = [];
              updateUIKeranjang();
              tampilkanStatus('success', 'Pembayaran Lunas!', `Link akses produk digital telah dikirim otomatis ke ${emailPembeli}. Cek Inbox/Spam email kamu.`);
            },
            onPending: function() {
              keranjang = [];
              updateUIKeranjang();
              tampilkanStatus('pending', 'Menunggu Pembayaran', `Segera selesaikan pembayaran. File otomatis terkirim ke ${emailPembeli} begitu lunas.`);
            },
            onError: function() {
              tampilkanStatus('error', 'Pembayaran Gagal', 'Silakan coba metode pembayaran lain.');
            }
          });
        } else {
          tampilkanStatus('error', 'Gagal Memproses', data.message || 'Cek kembali pengaturan Midtrans.');
        }
      } catch (err) {
        tampilkanStatus('error', 'Koneksi Bermasalah', 'Gagal terhubung ke server.');
      } finally {
        btn.disabled = false;
        btn.innerText = teksAsli;
      }
    }

    async function verifikasiPin() {
      const pin = document.getElementById('input-pin').value.trim();
      if (!pin) return;

      if (APPS_SCRIPT_URL.includes('GANTI_DENGAN_ID')) {
        if (pin !== '123456') {
          tutupModal('modal-pin');
          tampilkanStatus('error', 'PIN Salah', 'PIN default mode demo adalah 123456.');
          return;
        }
        adminPinTersimpan = pin;
        tutupModal('modal-pin');
        document.getElementById('stat-omzet').innerText = 'Rp 245.000 (Demo)';
        document.getElementById('stat-sukses').innerText = '6 Order';
        document.getElementById('stat-produk').innerText = katalogProduk.length + ' Item';
        renderDaftarAdmin();
        document.getElementById('modal-admin').classList.remove('hidden');
        return;
      }

      // Verifikasi PIN langsung ke server Apps Script & ambil statistik penjualan
      try {
        const res = await fetch(APPS_SCRIPT_URL, {
          method: 'POST',
          body: JSON.stringify({ action: 'getOwnerStats', pin: pin })
        });
        const json = await res.json();
        if (json.status === 'success') {
          adminPinTersimpan = pin;
          tutupModal('modal-pin');
          updateStatistikUI(json.stats);
          renderDaftarAdmin();
          document.getElementById('modal-admin').classList.remove('hidden');
        } else {
          tutupModal('modal-pin');
          tampilkanStatus('error', 'Akses Ditolak', json.message || 'PIN Pemilik Toko salah!');
        }
      } catch (err) {
        tutupModal('modal-pin');
        tampilkanStatus('error', 'Error', 'Gagal memverifikasi PIN ke server.');
      }
    }

    async function simpanProdukAdmin(e) {
      e.preventDefault();
      const btn = document.getElementById('btn-simpan-produk');
      btn.disabled = true;
      btn.innerText = 'Menyimpan...';

      const idLama = document.getElementById('adm-id').value;
      const dataProduk = {
        id_produk: idLama || 'DIG-' + Date.now().toString().slice(-5),
        nama_produk: document.getElementById('adm-nama').value,
        kategori: document.getElementById('adm-kategori').value,
        harga: Number(document.getElementById('adm-harga').value),
        gambar_url: document.getElementById('adm-gambar').value,
        preview_url: document.getElementById('adm-preview').value,
        isi_produk: document.getElementById('adm-isi').value,
        deskripsi: document.getElementById('adm-deskripsi').value,
        file_akses: document.getElementById('adm-file').value
      };

      try {
        if (APPS_SCRIPT_URL.includes('GANTI_DENGAN_ID')) {
          if (adminPinTersimpan !== '123456') {
            adminPinTersimpan = '';
            tutupModal('modal-admin');
            tampilkanStatus('error', 'PIN Salah', 'PIN default mode demo adalah 123456.');
            return;
          }
          if (idLama) {
            const idx = katalogProduk.findIndex(p => p.id_produk === idLama);
            if (idx > -1) katalogProduk[idx] = dataProduk;
          } else {
            katalogProduk.unshift(dataProduk);
          }
          resetFormAdmin();
          renderSemuaUI();
          return;
        }

        const res = await fetch(APPS_SCRIPT_URL, {
          method: 'POST',
          body: JSON.stringify({
            action: idLama ? 'updateProduct' : 'addProduct',
            pin: adminPinTersimpan,
            product: dataProduk
          })
        });
        const json = await res.json();
        if (json.status === 'success') {
          resetFormAdmin();
          await initToko();
        } else {
          adminPinTersimpan = '';
          tutupModal('modal-admin');
          tampilkanStatus('error', 'Akses Ditolak', json.message || 'PIN Admin tidak valid.');
        }
      } finally {
        btn.disabled = false;
        btn.innerText = 'Simpan Produk ke Katalog';
      }
    }

    async function hapusProdukAdmin(id) {
      try {
        if (APPS_SCRIPT_URL.includes('GANTI_DENGAN_ID')) {
          katalogProduk = katalogProduk.filter(p => p.id_produk !== id);
          renderSemuaUI();
          return;
        }
        const res = await fetch(APPS_SCRIPT_URL, {
          method: 'POST',
          body: JSON.stringify({ action: 'deleteProduct', pin: adminPinTersimpan, id_produk: id })
        });
        const json = await res.json();
        if (json.status === 'success') {
          await initToko();
        } else {
          tampilkanStatus('error', 'Gagal Menghapus', json.message);
        }
      } catch (err) {
        tampilkanStatus('error', 'Error', 'Gagal terhubung ke server.');
      }
    }

    function tampilkanStatus(tipe, judul, deskripsi) {
      const icon = document.getElementById('status-icon');
      if (tipe === 'success') {
        icon.className = 'w-16 h-16 rounded-full mx-auto flex items-center justify-center mb-4 bg-emerald-100 text-emerald-600 text-2xl font-bold';
        icon.innerText = '✓';
      } else if (tipe === 'pending' || tipe === 'info') {
        icon.className = 'w-16 h-16 rounded-full mx-auto flex items-center justify-center mb-4 bg-amber-100 text-amber-600 text-2xl font-bold';
        icon.innerText = '⚡';
      } else {
        icon.className = 'w-16 h-16 rounded-full mx-auto flex items-center justify-center mb-4 bg-rose-100 text-rose-600 text-2xl font-bold';
        icon.innerText = '!';
      }
      document.getElementById('status-title').innerText = judul;
      document.getElementById('status-desc').innerText = deskripsi;
      document.getElementById('modal-status').classList.remove('hidden');
    }

    function tutupModal(id) {
      document.getElementById(id).classList.add('hidden');
    }

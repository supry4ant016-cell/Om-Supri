        return Array.isArray(arr) && arr.length > 0 ? arr : [raw];

      if (!url) return null;
      const regExp = /(?:youtube\.com\/(?:watch\?v=|embed\/|shorts\/)|youtu\.be\/)([a-zA-Z0-9_-]{11})/;
      const match = url.match(regExp);

    }

      }
    }

    function syncSearch(val) {
      searchQuery = val.toLowerCase();
      document.getElementById('search-desktop').value = val;
      document.getElementById('search-mobile').value = val;
      filterKatalog();
    }

    function renderCategories() {
      const cats = ['ALL', ...new Set(allProducts.map(p => p.kategori).filter(Boolean))];
      document.getElementById('category-tabs').innerHTML = cats.map(c => `
        <button onclick="setKategori('${c}')" class="${c === activeCategory ? 'bg-zinc-900 text-white shadow-sm' : 'bg-white text-zinc-600 border border-zinc-200 hover:bg-zinc-50'} px-4 py-2 rounded-full text-xs font-extrabold shrink-0 transition">
          ${c === 'ALL' ? 'Semua Koleksi' : c}
        </button>
      `).join('');
    }

    function setKategori(cat) {
      activeCategory = cat;
      renderCategories();
      filterKatalog();
    }

    function filterKatalog() {
      const sortMode = document.getElementById('sort-select').value;
      let filtered = allProducts.filter(p => {
        const matchCat = activeCategory === 'ALL' || p.kategori === activeCategory;
        const matchText = p.nama_produk.toLowerCase().includes(searchQuery) || (p.deskripsi || '').toLowerCase().includes(searchQuery);
        return matchCat && matchText;
      });
      if (sortMode === 'low') filtered.sort((a, b) => Number(a.harga) - Number(b.harga));
      if (sortMode === 'high') filtered.sort((a, b) => Number(b.harga) - Number(a.harga));
      document.getElementById('product-count').innerText = filtered.length;
      renderProducts(filtered);
    }

    function bukaPreview(id) {
      const p = allProducts.find(x => x.id_produk === id);
      if (!p) return;
      const harga = Number(p.harga);

      document.getElementById('prev-cat').innerText = p.kategori;
      document.getElementById('prev-title').innerText = p.nama_produk;
      document.getElementById('prev-desc').innerText = p.deskripsi || '';
      document.getElementById('prev-old-price').innerText = 'Rp ' + Math.round(harga * 1.45).toLocaleString('id-ID');
      document.getElementById('prev-price').innerText = 'Rp ' + harga.toLocaleString('id-ID');

      // Setup Multi-Foto Portrait Carousel
      activeGallery = parseMediaList(p.media_list);
      activeGalleryIndex = 0;
      document.getElementById('photo-count-badge').innerText = activeGallery.length;
      renderGallery();

      // Setup Video Player (YouTube / MP4 Upload)
      const videoContainer = document.getElementById('video-player-container');
      const mediaTabs = document.getElementById('media-mode-tabs');
      if (p.video_url) {
        mediaTabs.classList.remove('hidden');
        const ytEmbed = getYouTubeEmbedUrl(p.video_url);
        if (ytEmbed) {
          videoContainer.innerHTML = `<iframe src="${ytEmbed}" class="w-full h-full" frameborder="0" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowfullscreen></iframe>`;
        } else {
          videoContainer.innerHTML = `<video src="${p.video_url}" controls playsinline class="w-full h-full object-contain"></video>`;
        }
        switchMediaTab('photos');
      } else {
        mediaTabs.classList.add('hidden');
        videoContainer.innerHTML = '';
        switchMediaTab('photos');
      }

      const items = (p.isi_produk || 'Akses penuh file produk digital').split('|');
      document.getElementById('prev-list').innerHTML = items.map(i => `<li class="flex items-start gap-2"><span class="text-emerald-500 font-extrabold">✓</span><span>${i.trim()}</span></li>`).join('');

      const sampleBtn = document.getElementById('prev-sample-btn');
      if (p.preview_url) {
        sampleBtn.href = p.preview_url;
        sampleBtn.classList.remove('hidden');
      } else {
        sampleBtn.classList.add('hidden');
      }

      document.getElementById('prev-add-cart').onclick = () => { tambahKeranjang(p.id_produk); tutupModal('modal-preview'); };
      document.getElementById('prev-buy-now').href = p.link_co || '#';
      document.getElementById('modal-preview').classList.remove('hidden');
    }

    function renderGallery() {
      document.getElementById('gallery-active-img').src = activeGallery[activeGalleryIndex];
      const prevBtn = document.getElementById('gal-prev-btn');
      const nextBtn = document.getElementById('gal-next-btn');
      const thumbsEl = document.getElementById('gallery-thumbs');

      if (activeGallery.length > 1) {
        prevBtn.classList.remove('hidden');
        nextBtn.classList.remove('hidden');
        thumbsEl.classList.remove('hidden');
        thumbsEl.innerHTML = activeGallery.map((img, idx) => `
          <img src="${img}" onclick="pilihFotoGaleri(${idx})" class="w-11 h-14 object-cover rounded-lg cursor-pointer border-2 transition ${idx === activeGalleryIndex ? 'border-indigo-600 scale-105' : 'border-transparent opacity-60 hover:opacity-100'}">
        `).join('');
      } else {
        prevBtn.classList.add('hidden');
        nextBtn.classList.add('hidden');
        thumbsEl.classList.add('hidden');
      }
    }

    function geserGaleri(dir) {
      activeGalleryIndex = (activeGalleryIndex + dir + activeGallery.length) % activeGallery.length;
      renderGallery();
    }

    function pilihFotoGaleri(idx) {
      activeGalleryIndex = idx;
      renderGallery();
    }

    function switchMediaTab(tab) {
      const gal = document.getElementById('gallery-container');
      const vid = document.getElementById('video-player-container');
      const btnP = document.getElementById('tab-btn-photos');
      const btnV = document.getElementById('tab-btn-video');

      if (tab === 'video') {
        gal.classList.add('hidden');
        vid.classList.remove('hidden');
        btnV.className = 'px-3.5 py-1.5 rounded-full text-xs font-extrabold bg-rose-600 text-white';
        btnP.className = 'px-3.5 py-1.5 rounded-full text-xs font-extrabold bg-zinc-100 text-zinc-700';
      } else {
        vid.classList.add('hidden');
        gal.classList.remove('hidden');
        btnP.className = 'px-3.5 py-1.5 rounded-full text-xs font-extrabold bg-zinc-900 text-white';
        btnV.className = 'px-3.5 py-1.5 rounded-full text-xs font-extrabold bg-zinc-100 text-zinc-700';
      }
    }

    function tambahKeranjang(id) {
      const p = allProducts.find(x => x.id_produk === id);
      if (!p) return;
      if (!cart.find(c => c.id_produk === id)) cart.push(p);
      updateCartUI();
    }

    function hapusKeranjang(id) {
      cart = cart.filter(c => c.id_produk !== id);
      updateCartUI();
      renderCartItems();
      if (cart.length === 0) tutupModal('modal-cart');
    }

    function updateCartUI() {
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      document.getElementById('nav-cart-count').innerText = cart.length;
      document.getElementById('cart-badge-mobile').innerText = cart.length;
      document.getElementById('cart-total-mobile').innerText = 'Rp ' + total.toLocaleString('id-ID');
      const mobBar = document.getElementById('floating-cart');
      if (cart.length > 0) mobBar.classList.remove('hidden');
      else mobBar.classList.add('hidden');
    }

    function bukaKeranjang() {
      if (cart.length === 0) return alert('Tas belanja masih kosong. Pilih produk terlebih dahulu!');
      renderCartItems();
      document.getElementById('modal-cart').classList.remove('hidden');
    }

    function renderCartItems() {
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);
      document.getElementById('cart-modal-total').innerText = 'Rp ' + total.toLocaleString('id-ID');
      document.getElementById('cart-items-list').innerHTML = cart.map(item => `
        <div class="flex items-center justify-between gap-2 bg-zinc-50 p-3 rounded-2xl border border-zinc-200/70">
          <div class="min-w-0 flex-1">
            <p class="text-xs font-bold text-zinc-800 truncate">${item.nama_produk}</p>
            <p class="text-xs font-extrabold text-indigo-600">Rp ${Number(item.harga).toLocaleString('id-ID')}</p>
          </div>
          <div class="flex items-center gap-1.5 shrink-0">
            <a href="${item.link_co || '#'}" target="_blank" class="bg-zinc-900 hover:bg-indigo-600 text-white px-3 py-1.5 rounded-xl text-[11px] font-extrabold">Bayar / CO &nearr;</a>
            <button onclick="hapusKeranjang('${item.id_produk}')" class="w-7 h-7 rounded-xl bg-rose-100 text-rose-600 text-xs font-bold">&times;</button>
          </div>
        </div>
      `).join('');
    }

    function checkoutViaWA() {
      const daftar = cart.map((item, idx) => `${idx + 1}. ${item.nama_produk} (Rp ${Number(item.harga).toLocaleString('id-ID')})`).join('%0A');
      const total = cart.reduce((sum, item) => sum + Number(item.harga), 0);

    }

    }
